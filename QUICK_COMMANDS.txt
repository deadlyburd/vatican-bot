cat backup.sql | docker-compose exec -T db psql -U postgres ticketbot

📦 BOKUN INTEGRATION COMMANDS
═══════════════════════════════════════════════════════════════════════════════

# Setup Bokun integration
docker-compose exec backend python /app/backend/manage.py setup_bokun

# Sync bookings now
docker-compose exec backend python /app/backend/manage.py setup_bokun --sync-now

# Test only (no changes)
docker-compose exec backend python /app/backend/manage.py setup_bokun --test-only

# Manual sync from Django shell
docker-compose exec backend python /app/backend/manage.py shell
>>> from backend.services.bokun_api import get_bokun_api
>>> api = get_bokun_api()
>>> bookings = api.get_confirmed_bookings()
>>> print(f"Found {len(bookings)} bookings")
>>> synced = api.sync_bookings_to_sheets()
>>> print(f"Synced {synced} bookings")
>>> exit()
